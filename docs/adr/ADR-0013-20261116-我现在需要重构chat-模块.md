## 二、 核心机制突破：基于对话树的时间旅行卡片状态机

在彻底重构中，**不要使用 `localStorage` 保存卡片状态**。

在 `assistant-ui` 中，对话是一棵支持分支（Branching）和编辑重发（Edit & Reload）的树。如果用户在第 1 轮表单点击“提交”生成了第 2 轮对话，随后又点击第 1 轮消息的“重新生成”或切换了历史分支，本地存储的状态就会与当前对话树彻底脱节。

**原生解法：让卡片状态完全由当前活跃分支的消息流确定性推导。**

```tsx
// src/components/chat/useBranchCardState.ts
"use client";

import { useMessage, useThread } from "@assistant-ui/react";

/**
 * 确定性推导当前 OpenUI 卡片的生命周期状态：
 * 1. 只要当前助手消息后面已经产生了后续消息，说明该轮交互已推进，卡片自动锁定（支持分支切换自动解锁）。
 * 2. 若紧随其后的用户消息包含结构化提交载荷，可直接解析回显用户当时的操作摘要。
 */
export function useBranchCardState() {
  const messageId = useMessage((m) => m.id);
  const isRunning = useMessage((m) => m.status?.type === "running");

  const { isCommitted, committedSummary } = useThread((t) => {
    const idx = t.messages.findIndex((m) => m.id === messageId);
    if (idx === -1) return { isCommitted: false, committedSummary: null };

    // 当前分支下，该条助手消息之后是否已有后续对话
    const nextMessage = t.messages[idx + 1];
    if (!nextMessage) {
      return { isCommitted: false, committedSummary: null };
    }

    // 提取下一条用户消息的文本作为已提交动作的回显依据
    const nextText = nextMessage.content
      .filter((p): p is { type: "text"; text: string } => p.type === "text")
      .map((p) => p.text)
      .join(" ");

    return {
      isCommitted: true,
      committedSummary: nextText || "已进入下一步操作",
    };
  });

  return {
    messageId,
    isRunning,
    isLocked: isRunning || isCommitted,
    committedSummary,
  };
}

```

**效果**：零外部存储依赖。当用户切换 `BranchPicker` 分支、或编辑某条历史提问重新生成时，旧分支上的表单和按钮会自动根据当前分支时间线解锁或锁定。

### 2. 流式性能调度渲染桥接（`src/components/chat/OpenUITextPart.tsx`）

```tsx
"use client";

import React, { useDeferredValue } from "react";
import { Renderer } from "@openuidev/react-lang";
import { openuiLibrary } from "@/lib/openui/registry";
import { useBranchCardState } from "./useBranchCardState";

// 仅用于防御大模型偶发输出不合法语法的运行时边界
class StreamSyntaxBoundary extends React.Component<
  { rawText: string; children: React.ReactNode },
  { hasError: boolean }
> {
  state = { hasError: false };
  static getDerivedStateFromError() {
    return { hasError: true };
  }
  render() {
    if (this.state.hasError) {
      return <div className="whitespace-pre-wrap text-sm">{this.props.rawText}</div>;
    }
    return this.props.children;
  }
}

export function OpenUITextPart({ text }: { text: string }) {
  const { messageId, isRunning } = useBranchCardState();

  // 流式输出期间降低 AST 解析优先级，确保输入打字与页面滚动 60fps 丝滑响应
  const deferredText = useDeferredValue(text);
  const displayText = isRunning ? deferredText : text;

  return (
    <StreamSyntaxBoundary key={messageId} rawText={text}>
      <div className="w-full overflow-x-auto">
        <Renderer
          response={displayText}
          library={openuiLibrary}
          isStreaming={isRunning}
        />
      </div>
    </StreamSyntaxBoundary>
  );
}
```

### 4. 后端统一接口（`src/app/api/chat/route.ts`）

```typescript
import { openai } from "@ai-sdk/openai";
import { streamText, convertToCoreMessages } from "ai";
import { generateSystemPrompt } from "@openuidev/lang-core";
import openuiSpec from "@/lib/openui/openui.spec.json";
import { sanitizeMessagesForLLM } from "@/lib/openui/sanitizer";

export const maxDuration = 60;

export async function POST(req: Request) {
  const { messages } = await req.json();

  const systemPrompt = generateSystemPrompt({
    library: openuiSpec,
    additionalInstructions: [
      "1. 遇到数据指标、对比分析、图表、表单或确认操作时，必须使用 OpenUI 组件输出。",
      "2. 用户上传附件（CSV/JSON/截图）后，优先提取核心维度并渲染为结构化 OpenUI 看板。",
      "3. 普通解释性文本直接输出简洁 Markdown。",
    ].join("\n"),
  });

  const coreMessages = convertToCoreMessages(messages);

  const result = streamText({
    model: openai("gpt-4o"),
    system: systemPrompt,
    messages: sanitizeMessagesForLLM(coreMessages),
    maxSteps: 5, // 后端静默完成所有业务 Tool 调用，最终一步统一输出 OpenUI Lang
  });

  return result.toDataStreamResponse();
}
```
